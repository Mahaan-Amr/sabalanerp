import os,stat,pathlib,json,sys,importlib.util
P=pathlib.Path
specfile,worktext=sys.argv[1:];work=P(worktext);spec=json.loads(P(specfile).read_text());accepted=spec['custody'];source=P(accepted['root'])
require=lambda v,c:None if v else (_ for _ in ()).throw(RuntimeError(c))
require(source.parent==work.parent and source.resolve(strict=True)==source and source.stat().st_uid==0 and not source.stat().st_mode&0o077,'SOURCE_ROOT_INVALID')
module=importlib.util.spec_from_file_location('private_images',work/'materialize.py');image=importlib.util.module_from_spec(module);module.loader.exec_module(image)
results={}
for service in ['backend','frontend','inquiry']:
 expected=accepted['decryption']['images'][service];raw=expected['verifiedRawArchive'];archive=source/(service+'.docker.tar.gz');s=archive.lstat()
 require(archive.resolve(strict=True)==archive and stat.S_ISREG(s.st_mode) and s.st_uid==0 and s.st_nlink==1 and not s.st_mode&0o077 and s.st_size==expected['plaintextBytes'],'SOURCE_ARCHIVE_INVALID')
 result=image.measure(archive,expected['plaintextSha256'],raw['index'],raw['platform'],raw['config'])
 require([v['diffId'] for v in result['layers']]==raw['rootfsDiffIds'],'SOURCE_DIFF_GRAPH_CHANGED');results[service]=result
proof={'format':'sabalan-independent-image-measurement-set/v1','images':results,'dockerImportVerified':False,'healthyApplicationPoint':False,'publicOnlineAdmission':False}
with (work/'image-measurement-proof.json').open('x') as f:os.fchmod(f.fileno(),0o400);json.dump(proof,f,indent=2);f.flush();os.fsync(f.fileno())
image.sync(work);print(json.dumps(proof))
